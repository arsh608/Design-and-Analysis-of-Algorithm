#include <iostream>
using namespace std;

int solve(int arr[], int low, int high)
{
    if (low == high)
    {
        return low;
    }

    int mid = low + (high - low) / 2;

    if (arr[mid] < arr[mid + 1])
    {
        return solve(arr, mid + 1, high);
    }
    else
    {
        return solve(arr, low, mid);
    }
}

int main()
{
     int n;
    cout << "Enter size of array = ";
    cin >> n;

    if (n < 10) {
        cout << "Size must be at least 10" << endl;
        return 1;
    }

    int* arr = new int[n];
    cout << "Enter " << n << " elements:" << endl;
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }

    int peak = solve(arr, 0, n - 1);

    cout << "Peak element found at index =  " << peak << " and  value = " << arr[peak] << endl;
    cout << "Time Complexity = O(n log n)" << endl;

    delete []arr;

    return 0;
}
