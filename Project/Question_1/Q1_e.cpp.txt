#include <iostream>
#include <algorithm>
#include <climits>
#include <cstdlib>
using namespace std;

int main() {
    int n;
    cout << "Enter size of array = ";
    cin >> n;

    if (n < 10) {
        cout << "Size must be at least 10" << endl;
        return 1;
    }

    int* arr = new int[n];
    cout << "Enter " << n << " elements:" << endl;
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }

    sort(arr, arr + n);

    int d = INT_MAX;
    for (int i = 0; i < n - 1; i++) {
        int temp = abs(arr[i + 1] - arr[i]);
        d = min(temp, d);
    }

    cout << "Closest distance = " << d << endl;
    cout << "Time Complexity = O(n log n)" << endl;

    delete[] arr;
    return 0;
}
