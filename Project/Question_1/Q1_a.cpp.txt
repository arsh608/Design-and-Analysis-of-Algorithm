#include <iostream>
using namespace std;

int count = 0; 

pair<int, int> solve(int arr[], int min, int max) {
    if (min == max) {
        return {arr[min], arr[min]};
    }

    if (max == min + 1) {
        count++;
        if (arr[min] < arr[max])
            return {arr[min], arr[max]};
        else
            return {arr[max], arr[min]};
    }

    int mid = (min + max) / 2;
    pair<int, int> left = solve(arr, min, mid);
    pair<int, int> right = solve(arr, mid + 1, max);

    count++;
    int total_min = (left.first < right.first) ? left.first : right.first;

    count++;
    int total_max = (left.second > right.second) ? left.second : right.second;

    return {total_min, total_max};
}

int main() {
    int n;
    cout << "Enter array size (n >= 10): ";
    cin >> n;

    if (n < 10) {
        cout << "Array size must be at least 10." << endl;
        return 0;
    }

    int* arr = new int[n];  

    cout << "Enter " << n << " elements" << endl;
    for (int i=0; i<n; i++) {
        cin >> arr[i];
    }

    pair<int, int> res = solve(arr, 0, n - 1);

    cout << "\nTotal key comparisons = " << count << endl;
     cout << "\nSmallest element = " << res.first;
    cout << "\nLargest element = " << res.second;
    cout << "Recurrence Relation = " << "T(n) = 2T(n/2) + 2  and solving  for n = 2^k gives T(n) = 2n - 2 key comparisons" << endl;
    cout << "Brute Force method takes 2(n - 1) comparisons while divide and conquer takes 2n - 2." << endl;
    cout << "Both are O(n), but divide and conquer uses slightly fewer comparisons theoretically." << endl;
   

    delete[] arr; 
    return 0;
}
