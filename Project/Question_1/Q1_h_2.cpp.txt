#include <iostream>
#include <vector>
using namespace std;
int mergeAndCount(vector<int> &a, int l, int m, int r)
{
    int cnt = 0;

    int j = m + 1;
    for (int i = l; i <= m; i++)
    {
        while (j <= r && a[i] > 2 * a[j])
            j++;
        cnt += (j - (m + 1));
    }

    vector<int> temp;
    int i = l;
    j = m + 1;
    while (i <= m && j <= r)
    {
        if (a[i] <= a[j])
            temp.push_back(a[i++]);
        else
            temp.push_back(a[j++]);
    }
    while (i <= m)
        temp.push_back(a[i++]);
    while (j <= r)
        temp.push_back(a[j++]);

    for (int k = l; k <= r; k++)
        a[k] = temp[k - l];

    return cnt;
}
int mergesortAndcount(vector<int> &a, int l, int r)
{
    if (l >= r)
        return 0;
    int m = (l + r) / 2;
    int cnt = 0;
    cnt += mergesortAndcount(a, l, m);
    cnt += mergesortAndcount(a, m + 1, r);
    cnt += mergeAndCount(a, l, m, r);
    return cnt;
}
int main()
{
    int n;
    cout << "Enter n: ";
    cin >> n;

    if (n < 10)
    {
        cout << "Size must be at least 10" << endl;
        return 1;
    }

    vector<int> a(n);
    cout << "Enter " << n << " values for array: ";
    for (int i = 0; i < n; i++)
    {
        cin >> a[i];
    }

    cout << "Significant Inversions: " << mergesortAndcount(a, 0, n - 1) << endl;
    return 0;
}
